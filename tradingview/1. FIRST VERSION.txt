stepSetup(Setup s, bool closedBar) =>
	bool alive = true
	bool bv    = false
	int  res   = 0
	int  b     = bar_index
	if s.state == ST_AWAY or s.state == ST_WAIT
		if closedBar and (s.dir < 0 ? close > s.sl : close < s.sl)
			s.state  := ST_INVALID
			s.endBar := b
			alive    := false
		else if s.state == ST_AWAY
			if (s.dir < 0 ? high < s.zl : low > s.zh)
				s.state := ST_WAIT
		else if (s.dir < 0 ? high >= s.zl : low <= s.zh)
			s.state := ST_VALID
			s.trig  := b
			s.outc  := OUT_OPEN
			s.risk  := math.max(math.abs(s.entry - s.sl), 1e-9)
			s.qHi   := high
			s.qLo   := low
			bv      := true
		else if b - s.brk > maxAge
			s.state  := ST_EXPIRED
			s.endBar := b
			alive    := false
	if alive and s.state == ST_VALID
		if b >= s.trig + 600
			alive := false
		else if (s.dir < 0 ? high >= s.sl : low <= s.sl)
			s.outc   := OUT_SL
			s.endBar := b
			alive    := false
			if not s.counted
				s.counted := true
				res       := -1
		else if b > s.trig
			s.maxR := math.max(s.maxR, (s.dir < 0 ? s.entry - low : high - s.entry) / s.risk)
			if s.h1 < 0 and s.maxR >= 1
				s.h1 := b
			if s.h2 < 0 and s.maxR >= 2
				s.h2 := b
			if s.h3 < 0 and s.maxR >= 3
				s.h3 := b
			if s.hTp < 0 and tpR0 > 0 and s.maxR >= tpR0
				s.hTp := b
			if s.hT < 0 and s.maxR >= tgt
				s.hT := b
				if not s.counted
					s.counted := true
					res       := 1
	[alive, bv, res]

wrapTxt(string s, int w) =>
	array<string> ws = str.split(s, " ")
	string out = ""
	int    cur = 0
	for wd in ws
		int l = str.length(wd)
		if cur > 0 and cur + 1 + l > w
			out += "\n" + wd
			cur := l
		else
			out += (cur > 0 ? " " : "") + wd
			cur += (cur > 0 ? 1 : 0) + l
	out
htfInfo() =>
	float tr_ = na(close[1]) ? high - low : math.max(high - low, math.abs(high - close[1]), math.abs(low - close[1]))
	var float a = na
	a := na(a) ? tr_ : (a * 13 + tr_) / 14
	int   m_  = math.max(3, minBars)
	float hh  = ta.highest(high, m_)
	float ll  = ta.lowest(low, m_)
	var int   md   = 0
	var int   p    = 1
	var float h    = na
	var float l    = na
	var float lim  = na
	var int   ld   = 0
	var int   cnt  = 0
	var int   sDir = 0
	var int   sSt  = -1
	var float szl  = na
	var float szh  = na
	var float ssl  = na
	var int   sBrk = 0
	int b = bar_index
	if sSt == 0 or sSt == 1
		if (sDir < 0 ? close > ssl : close < ssl)
			sSt := 3
		else if sSt == 0
			if (sDir < 0 ? high < szl : low > szh)
				sSt := 1
		else if (sDir < 0 ? high >= szl : low <= szh)
			sSt := 2
		else if b - sBrk > maxAge
			sSt := 3
	if md == 1
		if close > h or close < l
			int d = close > h ? 1 : -1
			cnt  := d == ld ? cnt + 1 : 1
			ld   := d
			szl  := d < 0 ? close : low
			szh  := d < 0 ? high : close
			ssl  := d < 0 ? math.max(h, high) + a * slBuf : math.min(l, low) - a * slBuf
			sDir := d
			sSt  := awayIn ? 0 : 1
			sBrk := b
			md   := 0
			p    := b + 1
		else
			float nh = math.max(h, high)
			float nl = math.min(l, low)
			if nh - nl > lim * 1.6
				md := 0
				p  := b
			else
				h := nh
				l := nl
	else if b - m_ + 1 >= p and hh - ll <= a * rangeAtr
		md  := 1
		h   := hh
		l   := ll
		lim := a * rangeAtr
	[ld[1], cnt[1], sDir[1], sSt[1]]
